var $magic: [int]int;

procedure helper();
  modifies $magic;
  ensures $magic == old($magic);
  ensures $magic == $magic[0 := 0];
  ensures $magic == $magic[1 := 2];

procedure p$main_2276();
  modifies $magic;

implementation p$main_2276()
{
  $magic[3] := 0;
  $magic[0] := 1;
  call helper();
  assert $magic[0] == 0;
  assert $magic[1] == 2;
  assert $magic[3] == 5;
  assert (forall #i: bv64 :: i > 1 <==> (if true then i > 1 else i <= 1));
}
